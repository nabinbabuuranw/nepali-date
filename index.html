For developers

Zero-dependency JavaScript library, MIT licensed. Supports 2000–2090 BS, every day tested.

import { adToBs, bsToAd, formatBs } from './src/nepali-date.js';

adToBs('2026-11-01');            // { year: 2083, month: 7, day: 15, weekday: 0 }
bsToAd(2083, 1, 1);              // { year: 2026, month: 4, day: 14, ... }
formatBs(adToBs(new Date()), 'YYYY MMMM D, dddd', 'ne'); // २०८३ कार्तिक १५, आइतबार
